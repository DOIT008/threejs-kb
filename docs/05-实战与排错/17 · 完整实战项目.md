# 完整实战项目

把前面学的串起来，做一个能跑的完整场景。把 `index.html` 的 `<script type="module">` 换成下面这段（用 r187 + importmap 环境）：

```js
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// ========== 1. 基础三件套 ==========
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x1a1a2e);
scene.fog = new THREE.Fog(0x1a1a2e, 12, 35);

const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 100);
camera.position.set(5, 4, 8);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFShadowMap;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
document.body.appendChild(renderer.domElement);

// ========== 2. 控制器 ==========
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.05;
controls.minDistance = 3;
controls.maxDistance = 20;
controls.maxPolarAngle = Math.PI / 2 - 0.05;
controls.target.set(0, 0.5, 0);
controls.autoRotate = true;
controls.autoRotateSpeed = 0.8;

// ========== 3. 光照 ==========
scene.add(new THREE.HemisphereLight(0x88bbff, 0x222233, 1.2));

const sun = new THREE.DirectionalLight(0xffffff, 3);
sun.position.set(5, 10, 5);
sun.castShadow = true;
sun.shadow.mapSize.set(2048, 2048);
sun.shadow.camera.left = -12;
sun.shadow.camera.right = 12;
sun.shadow.camera.top = 12;
sun.shadow.camera.bottom = -12;
scene.add(sun);

// ========== 4. 地面 ==========
const floor = new THREE.Mesh(
  new THREE.PlaneGeometry(40, 40),
  new THREE.MeshStandardMaterial({ color: 0x252540, roughness: 0.9 })
);
floor.rotation.x = -Math.PI / 2;
floor.receiveShadow = true;
scene.add(floor);

// ========== 5. 主角：发光球 ==========
const ball = new THREE.Mesh(
  new THREE.IcosahedronGeometry(1, 4),
  new THREE.MeshStandardMaterial({
    color: 0x42b883,
    roughness: 0.2,
    metalness: 0.6,
    emissive: 0x0a3d2a,
    emissiveIntensity: 0.5,
    flatShading: true
  })
);
ball.position.y = 1.2;
ball.castShadow = true;
scene.add(ball);

// ========== 6. 环绕的小方块（InstancedMesh） ==========
const cubeCount = 60;
const cubes = new THREE.InstancedMesh(
  new THREE.BoxGeometry(0.18, 0.18, 0.18),
  new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.4 }),
  cubeCount
);
cubes.castShadow = true;
const dummy = new THREE.Object3D();
const color = new THREE.Color();
for (let i = 0; i < cubeCount; i++) {
  dummy.position.set(
    (Math.random() - 0.5) * 20,
    Math.random() * 6,
    (Math.random() - 0.5) * 20
  );
  dummy.rotation.set(Math.random() * 3, Math.random() * 3, 0);
  dummy.updateMatrix();
  cubes.setMatrixAt(i, dummy.matrix);
  cubes.setColorAt(i, color.setHSL(Math.random(), 0.7, 0.6));
}
scene.add(cubes);

// ========== 7. 星空粒子 ==========
const starCount = 800;
const starPos = new Float32Array(starCount * 3);
for (let i = 0; i < starCount; i++) {
  starPos[i*3]     = (Math.random() - 0.5) * 60;
  starPos[i*3 + 1] = Math.random() * 30 + 5;
  starPos[i*3 + 2] = (Math.random() - 0.5) * 60;
}
const starGeo = new THREE.BufferGeometry();
starGeo.setAttribute('position', new THREE.BufferAttribute(starPos, 3));
const stars = new THREE.Points(starGeo, new THREE.PointsMaterial({
  color: 0xffffff, size: 0.15, transparent: true, opacity: 0.8
}));
scene.add(stars);

// ========== 8. 鼠标悬停交互 ==========
const raycaster = new THREE.Raycaster();
const mouse = new THREE.Vector2(-10, -10);
addEventListener('pointermove', e => {
  mouse.x = (e.clientX / innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / innerHeight) * 2 + 1;
});

// ========== 9. 动画循环（r187 新写法） ==========
const timer = new THREE.Timer();
renderer.setAnimationLoop(() => {
  timer.update();
  const t = timer.getElapsed();

  ball.position.y = 1.2 + Math.sin(t * 1.5) * 0.4;
  ball.rotation.y = t * 0.3;
  ball.rotation.x = t * 0.15;
  stars.rotation.y = t * 0.01;

  // 悬停检测
  raycaster.setFromCamera(mouse, camera);
  const hits = raycaster.intersectObject(ball);
  ball.material.emissiveIntensity = hits.length ? 1.5 : 0.5;
  document.body.style.cursor = hits.length ? 'pointer' : 'default';

  controls.update();
  renderer.render(scene, camera);
});

// ========== 10. 窗口自适应 ==========
addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
```

### 深一层：为什么这样搭——五层职责

会抄这段代码不算本事，能说清「为什么每段放在这、换成别行不行」才算真懂。这 10 个步骤其实分属五层：

| 层 | 包含步骤 | 职责 | 变化频率 |
| --- | --- | --- | --- |
| ① 环境层 | 1 三件套、10 自适应 | 提供「在哪画、画多大」 | 几乎不变 |
| ② 视角层 | 2 控制器 | 决定「从哪看」 | 偶尔调参 |
| ③ 内容层 | 3 光照、4 地面、5 主角、6 实例、7 星空 | 决定「画什么」 | 最常变 |
| ④ 交互层 | 8 射线、9 拾取逻辑 | 决定「用户操作怎么影响内容」 | 常变 |
| ⑤ 生命周期层 | 9 动画循环、销毁逻辑 | 决定「什么时候算、什么时候清」 | 项目稳定后不变 |

::: tip 大白话时间
把场景想成**一场舞台剧**：

- 环境层 = 剧场（舞台多大、观众坐哪）
- 视角层 = 摄像机机位
- 内容层 = 演员和道具
- 交互层 = 观众喊话时演员怎么回应
- 生命周期层 = 演出什么时候开始、什么时候谢幕收场

**分层的好处是：换演员不用拆剧场。** 你想把发光球换成 GLB 模型，只动内容层第 5 步；想加点击交互，只动交互层。不用整篇代码重排。
:::

**三个「为什么」：**

**为什么小方块用 `InstancedMesh` 而不是 `new Mesh` 60 次？**
60 个独立 Mesh = 60 次 draw call；一个 `InstancedMesh` = 1 次。这里只有 60 个，性能差别还不明显，但换成 6000 个就是天壤之别。代码结构提前按规模化写好，后面加量不用重写。

**为什么悬停检测放在动画循环里，而不是 `pointermove` 事件里？**
鼠标事件一秒可能触发上百次，每次都做射线检测是浪费。动画循环本来每秒就跑 60 次，在里面顺手做一次就够了。事件里只负责**记录鼠标坐标**，不做重活——这是交互层的核心分工。

**为什么 `controls.update()` 要写在 `renderer.render()` 前面？**
控制器改的是相机位置，`render()` 用的是当前相机状态。顺序反了，画面会慢一帧才跟上鼠标——人眼能感觉到那种「黏滞感」。

::: warning 这些代码看着独立，其实有依赖
- 第 5 步的主角用 `MeshStandardMaterial`，**必须有第 3 步的灯**才看得见
- 第 9 步的射线检测用 `camera`，**必须等第 2 步控制器更新完**才是准确的射线起点
- 第 10 步的 resize 同时改相机和渲染器，**漏一个就变形**

这就是为什么不能随便删代码——删之前先想清楚它属于哪层、被谁依赖。
:::

### 深一层：从「能跑」到「能交付」

上面这段代码粘到 `index.html` 就能看到效果，但**能跑 ≠ 能上线**。真实项目里还差下面这些。

**① 黑屏五步自查（按顺序查，命中率 95%）：**

| 步骤 | 检查什么 | 快速验证 |
| --- | --- | --- |
| 1 | 控制台有没有红字？ | 多数是 import 路径、WebGL 上下文创建失败 |
| 2 | 相机是不是看错了方向？ | 临时 `camera.position.set(0,0,10); camera.lookAt(0,0,0)` |
| 3 | 物体是不是在 near/far 之外？ | 确认 `near=0.1`、`far=100` 之类的合理值，物体在中间 |
| 4 | 有没有灯？材质是不是纯黑？ | 临时把材质换成 `MeshBasicMaterial({ color: 0xff0000 })`，能看见就是灯光问题 |
| 5 | canvas 有没有被 CSS 盖住/高度为 0？ | 检查 `renderer.domElement` 的 computed size |

::: tip 大白话时间
黑屏排查的核心思路是**二分法**：先换成最简单的材质、最简单的相机位置，确认「渲染管线通了」，再一块块加回来。

一上来就盯着自己那 200 行代码找 bug，是最慢的办法。
:::

**② 上线前 checklist：**

- [ ] 动画循环里调了 `controls.update()`（如果用阻尼/自动旋转）
- [ ] 限制 `renderer.setPixelRatio(Math.min(devicePixelRatio, 2))`
- [ ] 注册了 `resize` 监听，并更新 `camera.aspect` + `updateProjectionMatrix()`
- [ ] 组件销毁时清掉：事件监听、`controls.dispose()`、`geometry.dispose()`、`renderer.dispose()`
- [ ] 页面隐藏时暂停动画循环（`document.visibilityState`），省电省 GPU
- [ ] 移动端实测过：帧率、触摸手势、内存占用
- [ ] GLB / 贴图走了压缩（DRACO / KTX2），首屏资源控制在几 MB 内
- [ ] 有加载失败兜底和 WebGL 不支持的降级提示

**③ 在 Vue3 里怎么拆？**

不要把上面 147 行塞进一个组件。按职责拆成：

```text
src/
  composables/useThreeStage.js   // 三件套 + 动画循环 + 销毁
  composables/useDeviceData.js   // 业务数据（Pinia）
  components/Stage.vue           // 只管 canvas 容器
  components/DeviceMesh.vue      // 只管一个设备的几何体与 userData
  components/DetailPanel.vue     // 纯 DOM，不用 Three.js
```

核心原则：**Three.js 对象用 `shallowRef` / `markRaw` 包住，不要让 Vue 深度响应式接管**。具体写法见[「Vue3 集成案例」](</08-Vue3集成案例/Vue3-00 · 怎么用这些案例>)分组。
